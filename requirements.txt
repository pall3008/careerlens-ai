# CareerLens AI — deployment requirements
#
# This file is tuned for Streamlit Community Cloud (2 CPU / 2.7GB RAM ceiling).
# Voice input and RAG evaluation are LOCAL-ONLY extras — see the bottom of this
# file. The app degrades gracefully without them: the voice tab simply hides.

# ── CPU-only PyTorch ──────────────────────────────────────────────────────────
# Without this index, pip pulls the CUDA build on Linux (~2.5GB) and the
# deployment runs out of disk before it finishes installing.
--extra-index-url https://download.pytorch.org/whl/cpu
torch>=2.0.0

# ── Core UI ───────────────────────────────────────────────────────────────────
streamlit>=1.32.0

# ── LLM ───────────────────────────────────────────────────────────────────────
groq>=0.9.0                          # Groq SDK — calls Llama-3.3-70B (free, fast)

# ── RAG Framework ─────────────────────────────────────────────────────────────
llama-index-core>=0.14.0,<0.15.0
llama-index-embeddings-huggingface>=0.7.0,<0.8.0
llama-index-vector-stores-chroma>=0.4.0
llama-index-retrievers-bm25>=0.5.0

# ── Vector Store ──────────────────────────────────────────────────────────────
chromadb>=0.5.0
# Uncomment ONLY if the deployment logs show a sqlite3 version error from
# ChromaDB. app.py already contains the matching shim, guarded by try/except.
# pysqlite3-binary>=0.5.0

# ── Classical NLP / Retrieval ─────────────────────────────────────────────────
# Note: spaCy was removed deliberately. Skill extraction is a gazetteer over
# SKILLS_TAXONOMY in src/nlp_engine.py — no statistical model was ever used.
rank-bm25>=0.2.2

# ── Embeddings ────────────────────────────────────────────────────────────────
transformers>=4.50.0
sentence-transformers>=2.7.0
numpy>=1.24.0

# ── PDF Parsing ───────────────────────────────────────────────────────────────
PyMuPDF>=1.23.0

# ── Utilities ─────────────────────────────────────────────────────────────────
requests>=2.31.0
python-dotenv>=1.0.0

# ══════════════════════════════════════════════════════════════════════════════
# LOCAL-ONLY EXTRAS — do not add these to the deployed build.
# Install them in your local venv to run voice interview, the MCP server, or
# RAG evaluation:
#
#   pip install openai-whisper streamlit-mic-recorder soundfile fastmcp ragas
#
# Why they are not deployed:
#   openai-whisper + soundfile  → the base model alone exceeds the free RAM tier
#   streamlit-mic-recorder      → voice tab hides automatically when absent
#   fastmcp                     → separate process (python src/mcp_server.py)
#   ragas                       → offline evaluation, not part of the app
# ══════════════════════════════════════════════════════════════════════════════
